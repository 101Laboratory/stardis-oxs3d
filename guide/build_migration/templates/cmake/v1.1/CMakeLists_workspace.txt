# --------------------------------------------------------------------
# 5.2 模块加载辅助函数
# --------------------------------------------------------------------
function(load_modules module_list category_name)
    message(STATUS "")
    message(STATUS "--- ${category_name} ---")
    
    foreach(module_entry ${module_list})
        # 解析模块条目：path;display_name;requirement
        string(REPLACE ";" "|" module_entry_escaped "${module_entry}")
        string(REPLACE "|" ";" module_parts "${module_entry_escaped}")
        
        list(GET module_parts 0 module_path)
        list(GET module_parts 1 module_display)
        list(GET module_parts 2 module_requirement)
        
        # 检查模块是否存在
        set(cmake_file "${CMAKE_SOURCE_DIR}/${module_path}/CMakeLists.txt")
        
        if(EXISTS ${cmake_file})
            add_subdirectory(${module_path})
            message(STATUS "  [+] ${module_display}")
        else()
            if(module_requirement STREQUAL "REQUIRED")
                message(WARNING "  [!] ${module_display} not found (REQUIRED)")
            else()
                message(STATUS "  [ ] ${module_display} (not found, optional)")
            endif()
        endif()
    endforeach()
endfunction()
