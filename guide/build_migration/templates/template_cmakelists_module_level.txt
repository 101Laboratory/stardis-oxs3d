cmake_minimum_required(VERSION 3.25)

# 项目名从目录推导，允许上层覆盖
get_filename_component(MODULE_NAME ${CMAKE_CURRENT_SOURCE_DIR} NAME)
project(${MODULE_NAME} LANGUAGES C CXX)

# ---------- 项目类型配置 ----------
# 取值：LIB 或 EXE
set(PROJECT_TYPE LIB CACHE STRING "Project type: LIB or EXE")

# ---------- 源码收集（src 下所有文件） ----------
file(GLOB_RECURSE PROJECT_SOURCES
    CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h
)

# ---------- 定义目标 ----------
if(PROJECT_TYPE STREQUAL "EXE")
    add_executable(${MODULE_NAME} ${PROJECT_SOURCES})
    set(PROJECT_IS_EXECUTABLE TRUE)
else()
    add_library(${MODULE_NAME} STATIC ${PROJECT_SOURCES})
    set(PROJECT_IS_EXECUTABLE FALSE)
endif()

# ---------- 头文件可见性 ----------
target_include_directories(${MODULE_NAME}
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# ---------- 示例：声明第三方依赖（只 link，不处理路径） ----------
# target_link_libraries(${MODULE_NAME}
#     PRIVATE Random123_runtime
# )

# ---------- 运行时依赖部署（仅 EXE / Test） ----------
if(PROJECT_IS_EXECUTABLE)
    deploy_runtime_dependencies(${MODULE_NAME})
endif()

# ---------- 测试支持 ----------
option(ENABLE_TESTS "Enable tests" OFF)
if(ENABLE_TESTS)
    enable_testing()
    add_subdirectory(Tests)
endif()
